# Temperature in K, energy units, and standard concentration in M.
T=298.15
energy=kJ/mol
C0=1.0

# A and Ea select an Arrhenius rate for each forward step.
# The field A= is the prefactor, distinct from the species named A.
# dG_eq specifies the equilibrium free energy for the reversible step.
A + B <-> C ; A=1e7 ; Ea=40.0 ; dG_eq=-5.0
C -> D ; A=1e8 ; Ea=45.0
C -> E ; A=1e8 ; Ea=50.0

preset_set1 = {
  [A] = 1.0
  [B] = 1.0
  [C] = 0.0
  [D] = 0.0
  [E] = 0.0
}

# A plain number assigned to an energy uses the energy= unit; offsets declare their own units.
param energy_offset = 5.0; unit = kJ/mol

# Constrain the third activation energy using the second.
param Ea3 = Ea2 + energy_offset

# k2 and k3 are the rate constants calculated from A and Ea.
param concentration_floor = 1e-30; unit = M
let rate_ratio = k3 / k2
let fraction_D = [D] / max([D] + [E], concentration_floor)

# Volume is edited in litres and calculated in cubic metres.
param sample_volume = 1; unit = L
let product_amount = [D] * sample_volume; unit = mol
