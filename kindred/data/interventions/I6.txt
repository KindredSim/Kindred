# title: Triggered intermediate trapping
# Adds a trapping reagent when an intermediate accumulates past a threshold.
intervention_set1 = {
  [Precursor] = 1.0
  [Intermediate] = 0.0
  [Scavenger] = 0.0
  [Product] = 0.0
}

reaction: Precursor -> Intermediate; k=0.8
reaction: Intermediate + Scavenger -> Product; k=1.0
intervention: op=trigger; watch_species=Intermediate; threshold=0.25; direction=rising; max_count=1; min_interval=0.0; then={op=jump,species=Scavenger,delta=0.5}
